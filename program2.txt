static int set[20], nostate, noalpha, s, notransition, r, buffer[20];
char alphabet[20], c;
static int e_closure[20][20] = {0};
struct node* transition[20][20] = {NULL};
int closure_idx;

void main() {
    int i, j;

    printf("Enter the number of alphabets:\n");
    scanf("%d", &noalpha);
    getchar();

    printf("NOTE: [use letter 'e' as epsilon]\n");
    printf("NOTE: ['e' must be the last character, if it is present]\n");
    printf("\nEnter alphabets:\n");
    for (i = 0; i < noalpha; i++) {
        alphabet[i] = getchar();
        getchar();
    }

    printf("\nEnter the number of states:\n");
    scanf("%d", &nostate);

    printf("\nEnter the number of transitions:\n");
    scanf("%d", &notransition);
    
    printf("NOTE: [Transition is in the form -> q_from alphabet q_to]\n");
    printf("NOTE: [State numbers must be greater than zero]\n");
    printf("\nEnter transitions:\n");
    for (i = 0; i < notransition; i++) {
        scanf("%d %c%d", &r, &c, &s);
        insert_trantbl(r, c, s);
    }

    printf("\n");
    printf("e-closure of states\n");
    printf("--------------------\n");

    for (i = 1; i <= nostate; i++) {
        closure_idx = 0;
        for (j = 0; j < 20; j++) {
            buffer[j] = 0;
            e_closure[i][j] = 0;
        }
        findclosure(i, i);
        printf("e-closure(q%d): ", i);
        print_e_closure(i);
        printf("\n");
    }
}

void findclosure(int x, int sta) {
    struct node* temp;
    
    if (buffer[x]) {
        return;
    }

    e_closure[sta][closure_idx++] = x;
    buffer[x] = 1;

    if (alphabet[noalpha - 1] == 'e' && transition[x][noalpha - 1] != NULL) {
        temp = transition[x][noalpha - 1];
        while (temp != NULL) {
            findclosure(temp->st, sta);
            temp = temp->link;
        }
    }
}

void insert_trantbl(int r, char c, int s) {
    int j;
    struct node* temp;
    
    j = findalpha(c);
    if (j == 999) {
        printf("error: Alphabet not found\n");
        exit(0);
    }
    
    temp = (struct node*)malloc(sizeof(struct node));
    temp->st = s;
    temp->link = transition[r][j];
    transition[r][j] = temp;
}

int findalpha(char c) {
    int i;
    for (i = 0; i < noalpha; i++) {
        if (alphabet[i] == c)
            return i;
    }
    return (999);
}

void print_e_closure(int i) {
    int j;
    printf("{");
    for (j = 0; e_closure[i][j] != 0; j++) {
        printf("q%d,", e_closure[i][j]);
    }
    if (e_closure[i][0] != 0) {
        printf("\b");
    }
    printf("}");
}
